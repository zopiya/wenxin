{{- /* 中文农历日期格式化（非中文站点语言时回退为标准公历日期） 用法： 仅月日：{{ partial "chinese-date.html" .Date }} 含年份：{{ partial "chinese-date.html" (dict "date" .Date "year" true "site" .Site) }} 输出（中文站点，site.Language.Lang 以 "zh" 开头）： 仅月日：正月初一 含年份：乙巳年正月初一 输出（非中文站点，需传入 "site" 才能生效）： 标准 Gregorian 日期，按 :date_medium / :date_long 本地化格式 */ -}} {{- $date := . -}} {{- $showYear := false -}} {{- $lang := "zh" -}} {{- if reflect.IsMap . -}} {{- $date = .date -}} {{- $showYear = .year | default false -}} {{- with .site -}} {{- $lang = .Language.Lang -}} {{- end -}} {{- end -}} {{- $result := "" -}} {{- /* Lunar (天干地支) formatting is a Chinese-locale feature; non-Chinese site languages fall back to a standard, i18n-aware Gregorian date so the theme stays usable for non-Chinese content. */ -}} {{- if not (strings.HasPrefix $lang "zh") -}} {{- if $showYear -}} {{- $result = $date | time.Format ":date_long" -}} {{- else -}} {{- $result = $date | time.Format ":date_medium" -}} {{- end -}} {{- else -}} {{- $dateStr := $date.Format "20060102" -}} {{- $entry := index hugo.Data.chinese_date.calendar $dateStr -}} {{- if $entry -}} {{- $lunarYear := int (index $entry 0) -}} {{- $monthIdx := int (index $entry 1) -}} {{- $dayIdx := int (index $entry 2) -}} {{- $isLeap := int (index $entry 3) -}} {{- $monthName := "" -}} {{- if eq $isLeap 1 -}} {{- $monthName = index hugo.Data.chinese_date.leap_months $monthIdx -}} {{- else -}} {{- $monthName = index hugo.Data.chinese_date.months $monthIdx -}} {{- end -}} {{- $dayName := index hugo.Data.chinese_date.days $dayIdx -}} {{- if $showYear -}} {{- $yearName := index hugo.Data.chinese_date.years (printf "%d" $lunarYear) -}} {{- $result = printf "%s%s%s" $yearName $monthName $dayName -}} {{- else -}} {{- $result = printf "%s%s" $monthName $dayName -}} {{- end -}} {{- else -}} {{- $monthNames := slice "一" "二" "三" "四" "五" "六" "七" "八" "九" "十" "十一" "十二" -}} {{- $monthPart := printf "%s月" (index $monthNames (sub (int $date.Month) 1)) -}} {{- $day := $date.Day -}} {{- $digits := slice "零" "一" "二" "三" "四" "五" "六" "七" "八" "九" "十" -}} {{- $dayPart := "" -}} {{- if le $day 10 -}} {{- $dayPart = printf "初%s日" (index $digits $day) -}} {{- else if le $day 20 -}} {{- if eq $day 20 -}} {{- $dayPart = "二十日" -}} {{- else -}} {{- $dayPart = printf "十%s日" (index $digits (sub $day 10)) -}} {{- end -}} {{- else if lt $day 30 -}} {{- $dayPart = printf "廿%s日" (index $digits (sub $day 20)) -}} {{- else if eq $day 30 -}} {{- $dayPart = "三十日" -}} {{- else -}} {{- $dayPart = "三十一日" -}} {{- end -}} {{- $result = printf "%s%s" $monthPart $dayPart -}} {{- end -}} {{- end -}} {{- $result -}}
